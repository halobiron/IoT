# Window 1: receives ESP32 sensor data every 2 seconds.
mosquitto_sub -V mqttv311 -h 2a2b78e5fa66468b9ff212a02f7c991d.s1.eu.hivemq.cloud -p 8883 --cafile "isrgrootx1.pem" -u "TranHaiLong" -P "B23DCCN510" -t "esp32/iot/data"

# Window 2: receives ESP32's LED status response after each command.
mosquitto_sub -V mqttv311 -h 2a2b78e5fa66468b9ff212a02f7c991d.s1.eu.hivemq.cloud -p 8883 --cafile "isrgrootx1.pem" -u "TranHaiLong" -P "B23DCCN510" -t "esp32/iot/action-history"

# Window 3: sends a request to the ESP32.
mosquitto_pub -V mqttv311 -h 2a2b78e5fa66468b9ff212a02f7c991d.s1.eu.hivemq.cloud -p 8883 --cafile "isrgrootx1.pem" -u "TranHaiLong" -P "B23DCCN510" -t "esp32/iot/control" -m "led1:on, led2:off"
